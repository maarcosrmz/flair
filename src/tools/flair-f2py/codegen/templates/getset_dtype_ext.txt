    ! ===== {field} property (derived-type view, type wrapped in another file) =====

    ! getter: the view object is allocated by the defining module's wrapper via
    ! the external view converter; its ptr aliases p%{field} inside this parent's
    ! heap object and holds a reference on the parent (numpy 'base' pattern).
    ! On failure the converter has set the exception and returned null.
    function {get_fn}(self, closure) bind(C) result(r)
        type(c_ptr), value :: self, closure
        type(c_ptr) :: r
        type({tname}), pointer :: p
        call c_f_pointer(FLAIR_data_ptr(self), p)
        r = {view_fn}(c_loc(p%{field}), self)
    end function

    ! setter: copy-in through the external unwrap converter (isinstance-checked
    ! there; it sets the exception and returns a disassociated pointer on
    ! failure). Intrinsic assignment deep-copies allocatable components.
    function {set_fn}(self, value, closure) bind(C) result(r)
        type(c_ptr), value :: self, value, closure
        integer(c_int) :: r
        type({tname}), pointer :: p
        type({sub_tname}), pointer :: v
        r = -1
        if (.not. c_associated(value)) then
            call PyErr_SetString(PyExc_TypeError, c_loc({s_del}))
            return
        end if
        v => {from_fn}(value)
        if (.not. associated(v)) return
        call c_f_pointer(FLAIR_data_ptr(self), p)
        ! aliased self-assignment (value is a view of this very component) is not
        ! reliably temped by compilers when the LHS has allocatable components
        if (.not. associated(v, p%{field})) then
            p%{field} = v
        end if
        r = 0
    end function
