    ! ===== {field} property (derived-type view) =====

    ! getter: returns a view object whose data pointer aliases p%{field} inside the
    ! parent's heap object. The view holds a reference on the parent (numpy 'base'
    ! pattern), so the parent outlives every view; dealloc of a view only drops
    ! that reference and never deallocates.
    function {get_fn}(self, closure) bind(C) result(r)
        type(c_ptr), value :: self, closure
        type(c_ptr) :: r
        type({tname}), pointer :: p
        call c_f_pointer(FLAIR_data_ptr(self), p)
        r = FLAIR_new_view({type_obj}, c_loc(p%{field}), self)
    end function

    ! setter: copy-in. Requires a {sub_tname} instance; intrinsic assignment
    ! deep-copies allocatable components.
    function {set_fn}(self, value, closure) bind(C) result(r)
        type(c_ptr), value :: self, value, closure
        integer(c_int) :: r
        type({tname}), pointer :: p
        type({sub_tname}), pointer :: v
        type(c_ptr) :: raw
        r = -1
        if (.not. c_associated(value)) then
            call PyErr_SetString(PyExc_TypeError, c_loc({s_del}))
            return
        end if
        raw = FLAIR_unwrap(value, {type_obj}, c_loc({s_type}))
        if (.not. c_associated(raw)) return
        call c_f_pointer(FLAIR_data_ptr(self), p)
        call c_f_pointer(raw, v)
        ! aliased self-assignment (value is a view of this very component) is not
        ! reliably temped by compilers when the LHS has allocatable components
        if (.not. associated(v, p%{field})) then
            p%{field} = v
        end if
        r = 0
    end function
