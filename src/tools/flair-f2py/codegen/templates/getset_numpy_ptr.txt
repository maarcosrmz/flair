    ! ===== {field} property -- NumPy array copy of p%{field} (rank 1, pointer) =====

    ! getter: returns a freshly-allocated, NumPy-owned copy of p%{field}.
    ! p%{field} is a pointer and may alias a strided (non-contiguous) target; the
    ! copy is taken via Fortran whole-array assignment, which honours the source
    ! layout, so any strided target is flattened to a plain contiguous result. NumPy
    ! owns the buffer (NPY_ARRAY_OWNDATA), so the result is fully independent of the
    ! Fortran instance -- no base object, no aliasing.
    function {get_fn}(self, closure) bind(C) result(r)
        type(c_ptr), value :: self, closure
        type(c_ptr) :: r, descr
        type({struct}), pointer :: pt
        type({tname}), pointer :: p
        integer(c_ptrdiff_t), target :: dims_arr(1)
        integer(c_ptrdiff_t) :: n
        {raw_type}, pointer :: raw(:)

        r = c_null_ptr
        if (.not. c_associated(numpy_api_ptr)) then
            call PyErr_SetString(PyExc_RuntimeError, c_loc(s_numpy_req))
            return
        end if

        call c_f_pointer(self, pt)
        call c_f_pointer(pt%{ptr_field}, p)

        if (associated(p%{field})) then
            n = int(size(p%{field}), c_ptrdiff_t)
        else
            n = 0_c_ptrdiff_t
        end if

        dims_arr(1) = n
        descr = PyArray_DescrFromType({npy})
        r = PyArray_NewFromDescr(PyArray_Type_ptr, descr, 1_c_int, &
                c_loc(dims_arr), c_null_ptr, c_null_ptr, &
                NPY_ARRAY_BEHAVED + NPY_ARRAY_OWNDATA, c_null_ptr)
        if (.not. c_associated(r)) return
        if (n > 0) then
            call c_f_pointer(PyArray_DATA(r), raw, [n])
            raw = p%{field}          ! whole-array copy honours the (possibly strided) source layout
        end if
    end function

    ! setter: write through the associated target (no realloc). Requires the pointer
    ! to be associated and the incoming size to match; copies element-wise honouring
    ! the incoming (possibly negative) stride. Intrinsic assignment to a pointer
    ! writes into its target, so a strided target is updated in place.
    function {set_fn}(self, value, closure) bind(C) result(r)
        type(c_ptr), value :: self, value, closure
        integer(c_int) :: r
        type({struct}), pointer :: pt
        type({tname}), pointer :: p
        type(c_ptr) :: arr, descr, data_ptr
        integer(c_ptrdiff_t) :: n, stride_bytes, stride_elem, total_span, base_addr
        {raw_type}, pointer :: raw(:)

        r = -1
        if (.not. c_associated(value)) then
            call PyErr_SetString(PyExc_TypeError, c_loc({s_del}))
            return
        end if
        if (.not. c_associated(numpy_api_ptr)) then
            call PyErr_SetString(PyExc_RuntimeError, c_loc(s_numpy_req))
            return
        end if

        call c_f_pointer(self, pt)
        call c_f_pointer(pt%{ptr_field}, p)
        if (.not. associated(p%{field})) then
            call PyErr_SetString(PyExc_RuntimeError, c_loc({s_unassoc}))
            return
        end if

        descr = PyArray_DescrFromType({npy})
        arr = PyArray_FromAny(value, descr, 1, 1, 0, c_null_ptr)
        if (.not. c_associated(arr)) return

        n            = PyArray_DIM(arr, 0_c_int)
        stride_bytes = PyArray_STRIDE(arr, 0_c_int)
        stride_elem  = stride_bytes / {elem_bytes}_c_ptrdiff_t  ! signed stride in elements
        data_ptr     = PyArray_DATA(arr)

        if (n /= int(size(p%{field}), c_ptrdiff_t)) then
            call Py_DecRef(arr)
            call PyErr_SetString(PyExc_ValueError, c_loc({s_size}))
            return
        end if

        if (n > 0) then
            total_span = 1_c_ptrdiff_t + (n - 1_c_ptrdiff_t) * abs(stride_elem)
            if (stride_elem > 0) then
                call c_f_pointer(data_ptr, raw, [total_span])
                p%{field} = raw(1:total_span:stride_elem)
            else
                base_addr = transfer(data_ptr, 0_c_ptrdiff_t) &
                          + (n - 1_c_ptrdiff_t) * stride_bytes
                call c_f_pointer(transfer(base_addr, c_null_ptr), raw, [total_span])
                p%{field} = raw(total_span:1:stride_elem)
            end if
        end if
        call Py_DecRef(arr)
        r = 0
    end function
