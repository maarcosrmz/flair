    ! ===== {field} property -- NumPy array copy of p%{field} (rank 1, pointer) =====

    ! getter: a freshly-allocated, NumPy-owned copy of p%{field}. The pointer
    ! may alias a strided target; the copy goes through Fortran whole-array
    ! assignment, which honours the source layout and flattens it into the
    ! contiguous NumPy buffer. The result is independent of the Fortran
    ! instance -- no base object and no aliasing.
    function {get_fn}(self, closure) bind(C) result(r)
        type(c_ptr), value :: self, closure
        type(c_ptr) :: r, buf
        type({tname}), pointer :: p
        integer(c_ptrdiff_t) :: dims(1)
        {raw_type}, pointer :: raw(:)

        call c_f_pointer(FLAIR_data_ptr(self), p)
        dims(1) = 0_c_ptrdiff_t
        if (associated(p%{field})) dims(1) = int(size(p%{field}), c_ptrdiff_t)

        r = FLAIR_array_new({npy}, 1_c_int, dims, buf)
        if (.not. c_associated(r)) return
        if (dims(1) > 0) then
            call c_f_pointer(buf, raw, dims)
            raw = p%{field}
        end if
    end function

    ! setter: write through the associated target rather than reallocating, so
    ! the pointer keeps pointing where it did and the incoming size must match.
    ! Assignment to a pointer writes into its target, so a strided target is
    ! updated in place; the coercion has already flattened the incoming side.
    function {set_fn}(self, value, closure) bind(C) result(r)
        type(c_ptr), value :: self, value, closure
        integer(c_int) :: r
        type({tname}), pointer :: p
        type(c_ptr) :: arr
        integer(c_ptrdiff_t) :: dims(1)
        {raw_type}, pointer :: raw(:)

        r = -1
        if (.not. c_associated(value)) then
            call PyErr_SetString(PyExc_TypeError, c_loc({s_del}))
            return
        end if

        call c_f_pointer(FLAIR_data_ptr(self), p)
        if (.not. associated(p%{field})) then
            call PyErr_SetString(PyExc_RuntimeError, c_loc({s_unassoc}))
            return
        end if

        arr = FLAIR_array_from_PyObject(value, {npy}, 1_c_int, .false., dims)
        if (.not. c_associated(arr)) return

        if (dims(1) /= int(size(p%{field}), c_ptrdiff_t)) then
            call FLAIR_array_release(arr)
            call PyErr_SetString(PyExc_ValueError, c_loc({s_size}))
            return
        end if

        if (dims(1) > 0) then
            call c_f_pointer(PyArray_DATA(arr), raw, dims)
            p%{field} = raw
        end if
        call FLAIR_array_release(arr)
        r = 0
    end function
