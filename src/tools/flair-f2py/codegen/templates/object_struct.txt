    ! Instance layout: stable ABI -- PyObject_HEAD is opaque (sizeof(PyObject)=16 on 64-bit).
    type, bind(C) :: {struct}

        type(c_ptr)       :: owner         ! non-null: view; PyObject that owns the storage (holds a ref)
