    ! ===== {field} property -- NumPy array copy of p%{field} (rank 1) =====

    ! getter: a freshly-allocated, NumPy-owned copy of p%{field}, so the result
    ! is fully independent of the Fortran instance -- no base object and no
    ! aliasing. The copy goes through Fortran array assignment into the
    ! contiguous NumPy buffer, which assumes nothing about the source layout.
    function {get_fn}(self, closure) bind(C) result(r)
        type(c_ptr), value :: self, closure
        type(c_ptr) :: r, buf
        type({tname}), pointer :: p
        integer(c_ptrdiff_t) :: dims(1)
        {raw_type}, pointer :: raw(:)

        call c_f_pointer(FLAIR_data_ptr(self), p)
        dims(1) = 0_c_ptrdiff_t
        if (allocated(p%{field})) dims(1) = int(size(p%{field}), c_ptrdiff_t)

        r = FLAIR_array_new({npy}, 1_c_int, dims, buf)
        if (.not. c_associated(r)) return
        if (dims(1) > 0) then
            call c_f_pointer(buf, raw, dims)
            raw = p%{field}
        end if
    end function

    ! setter: coerce to a contiguous {npy} array and copy into a fresh
    ! allocatable. The coercion materialises any strided or reversed input, so
    ! what remains here is a plain whole-array assignment.
    function {set_fn}(self, value, closure) bind(C) result(r)
        type(c_ptr), value :: self, value, closure
        integer(c_int) :: r
        type({tname}), pointer :: p
        type(c_ptr) :: arr
        integer(c_ptrdiff_t) :: dims(1)
        {raw_type}, pointer :: raw(:)

        r = -1
        if (.not. c_associated(value)) then
            call PyErr_SetString(PyExc_TypeError, c_loc({s_del}))
            return
        end if

        arr = FLAIR_array_from_PyObject(value, {npy}, 1_c_int, .false., dims)
        if (.not. c_associated(arr)) return

        call c_f_pointer(FLAIR_data_ptr(self), p)
        if (allocated(p%{field})) deallocate(p%{field})
        if (dims(1) > 0) then
            allocate(p%{field}(dims(1)))
            call c_f_pointer(PyArray_DATA(arr), raw, dims)
            p%{field} = raw
        end if
        call FLAIR_array_release(arr)
        r = 0
    end function
