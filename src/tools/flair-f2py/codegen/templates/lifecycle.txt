    ! ===== Lifecycle: {tname} =====

    function {new_fn}(type_ptr, args, kwds) bind(C) result(r)
        type(c_ptr), value :: type_ptr, args, kwds
        type(c_ptr) :: r
        type(FLAIR_object_t), pointer :: pt
        type({tname}), pointer :: p
{new_body}
    end function

    function {init_fn}(self, args, kwds) bind(C) result(r)
        type(c_ptr), value :: self, args, kwds
        integer(c_int) :: r
{init_body}
    end function

    subroutine {dealloc_fn}(self) bind(C)
        type(c_ptr), value :: self
        type({tname}), pointer :: p
        type(c_ptr) :: owned

        ! non-null only when this instance owns its storage, i.e. is not a view
        owned = FLAIR_dealloc_data(self)
        if (c_associated(owned)) then
            call c_f_pointer(owned, p)
            deallocate(p)
        end if
        call PyObject_Free(self)
    end subroutine
